--- Query_4_AGGREGATION ---
SQL:
SELECT
            c.category_name,
            COUNT(b.book_id) AS total_books,
            ROUND(AVG(b.price_gbp), 2) AS average_price_gbp
        FROM categories c
        JOIN books b
            ON c.category_id = b.category_id
        GROUP BY c.category_id, c.category_name
        ORDER BY total_books DESC;

--- Query_5_PK_FK_JOIN ---
SQL:
SELECT
            b.title,
            b.price_inr,
            b.rating,
            c.category_name
        FROM books b
        JOIN categories c
            ON b.category_id = c.category_id
        ORDER BY
            b.rating DESC,
            b.price_inr ASC
